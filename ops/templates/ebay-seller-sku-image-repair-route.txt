const STORE_ID = "00000000-0000-4000-8000-000000000001";
const TARGET_IDS = [1682,1692,1709,1723,1779,1863,1976,1983,1988,1992,2003,2019,2036,2052,2094,2103,2117,2183,2210,2211,2213,2228,2240,2241,2260,2273,2286,2291,2313,2315,2336,2366,2373,2386,2391,2411,2413,2419,2448,10650,10658,10664,10665,10666,10684,10685,10688,10689,10692,10699];
const PAGE_SIZE = 100;
const CONCURRENCY = 8;
const MAX_IMAGES = 20;

type ProductRow = { id: number; title: string; ebay_item_id: string | null };
type InventoryRow = { id: string; legacy_product_id: number; title: string };
type ImageRow = { id: string; inventory_item_id: string; image_url: string; sort_order: number; is_primary: boolean };
type SellerListing = { itemId: string; sku: string; pictureUrls: string[] };

function safeEqual(a: string, b: string) { const x = Buffer.from(a); const y = Buffer.from(b); return x.length === y.length && timingSafeEqual(x, y); }
function decodeXml(value: string) { return value.trim().replace(/^<!\[CDATA\[/, "").replace(/\]\]>$/, "").replace(/&#x([0-9a-f]+);/gi, (_, h) => String.fromCodePoint(Number.parseInt(h, 16))).replace(/&#([0-9]+);/g, (_, d) => String.fromCodePoint(Number.parseInt(d, 10))).replace(/&quot;/g, '"').replace(/&apos;/g, "'").replace(/&lt;/g, "<").replace(/&gt;/g, ">").replace(/&amp;/g, "&"); }
function xmlBlock(xml: string, tag: string) { return new RegExp(`<${tag}(?:\\s[^>]*)?>([\\s\\S]*?)<\\/${tag}>`, "i").exec(xml)?.[1] || null; }
function xmlBlocks(xml: string, tag: string) { return Array.from(xml.matchAll(new RegExp(`<${tag}(?:\\s[^>]*)?>([\\s\\S]*?)<\\/${tag}>`, "gi")), m => m[1]); }
function xmlText(xml: string, tag: string) { const block = xmlBlock(xml, tag); return block === null ? null : decodeXml(block); }
function hi(value: unknown) { const text = String(value || "").trim(); return text ? text.replace(/\/s-l\d+\.(jpg|jpeg|png|webp)(\?.*)?$/i, "/s-l1600.$1$2") : ""; }
function identity(value: unknown) { const text = hi(value); if (!text) return ""; try { const url = new URL(text); if (url.hostname.toLowerCase() === "i.ebayimg.com") { const modern = /^\/images\/g\/([^/]+)\//i.exec(url.pathname)?.[1]; const legacy = /\/z\/([^/]+)\/\$_\d+\.(?:jpe?g|png|webp)$/i.exec(url.pathname)?.[1]; if (modern || legacy) return `ebay:${modern || legacy}`; } url.hash = ""; return url.toString(); } catch { return text; } }
function normalize(values: unknown[]) { const result: string[] = []; const seen = new Set<string>(); for (const value of values) { const url = hi(value); const key = identity(url); if (!url || !key || seen.has(key)) continue; seen.add(key); result.push(url); if (result.length >= MAX_IMAGES) break; } return result; }
async function workers<T>(items: T[], worker: (item: T) => Promise<void>) { let cursor = 0; async function run() { while (cursor < items.length) await worker(items[cursor++]); } await Promise.all(Array.from({ length: Math.min(CONCURRENCY, Math.max(items.length, 1)) }, () => run())); }

async function accessToken(supabase: ReturnType<typeof createSupabaseServerClient>) {
  const clientId = process.env.EBAY_CLIENT_ID; const clientSecret = process.env.EBAY_CLIENT_SECRET;
  if (!clientId || !clientSecret) throw new Error("Missing eBay credentials");
  const { data, error } = await supabase.from("ebay_tokens").select("refresh_token").eq("store_id", STORE_ID).order("created_at", { ascending: false }).limit(1).maybeSingle();
  if (error) throw error; if (!data?.refresh_token) throw new Error("Missing eBay refresh token");
  const response = await fetch("https://api.ebay.com/identity/v1/oauth2/token", { method: "POST", headers: { Authorization: `Basic ${Buffer.from(`${clientId}:${clientSecret}`).toString("base64")}`, "Content-Type": "application/x-www-form-urlencoded" }, body: new URLSearchParams({ grant_type: "refresh_token", refresh_token: String(data.refresh_token), scope: "https://api.ebay.com/oauth/api_scope https://api.ebay.com/oauth/api_scope/sell.inventory" }), signal: AbortSignal.timeout(20000) });
  const payload = await response.json().catch(() => ({})); if (!response.ok || !payload.access_token) throw new Error(payload.error_description || payload.error || "Token refresh failed"); return String(payload.access_token);
}

async function sellerPage(token: string, page: number, from: string, to: string) {
  const response = await fetch("https://api.ebay.com/ws/api.dll", { method: "POST", headers: { "Content-Type": "text/xml", "X-EBAY-API-CALL-NAME": "GetSellerList", "X-EBAY-API-COMPATIBILITY-LEVEL": "1409", "X-EBAY-API-SITEID": "0", "X-EBAY-API-IAF-TOKEN": token }, body: `<?xml version="1.0" encoding="utf-8"?><GetSellerListRequest xmlns="urn:ebay:apis:eBLBaseComponents"><DetailLevel>ReturnAll</DetailLevel><EndTimeFrom>${from}</EndTimeFrom><EndTimeTo>${to}</EndTimeTo><IncludeVariations>false</IncludeVariations><Pagination><EntriesPerPage>${PAGE_SIZE}</EntriesPerPage><PageNumber>${page}</PageNumber></Pagination></GetSellerListRequest>`, signal: AbortSignal.timeout(60000) });
  const xml = await response.text(); const ack = xmlText(xml, "Ack") || "Failure"; if (!response.ok || !["Success", "Warning"].includes(ack)) throw new Error(xmlText(xmlBlock(xml, "Errors") || xml, "LongMessage") || `GetSellerList ${response.status}`);
  const items = xmlBlocks(xmlBlock(xml, "ItemArray") || "", "Item").flatMap(item => { const itemId = xmlText(item, "ItemID")?.trim(); if (!itemId) return []; const details = xmlBlock(item, "PictureDetails") || ""; return [{ itemId, sku: xmlText(item, "SKU")?.trim() || "", pictureUrls: normalize([...xmlBlocks(details, "PictureURL").map(decodeXml), xmlText(details, "GalleryURL")]) } satisfies SellerListing]; });
  const pagination = xmlBlock(xml, "PaginationResult") || ""; return { items, totalPages: Math.max(Number(xmlText(pagination, "TotalNumberOfPages") || 1), 1) };
}

async function allSellerListings(token: string) {
  const now = Date.now(); const from = new Date(now - 5 * 60 * 1000).toISOString(); const to = new Date(now + 119 * 24 * 60 * 60 * 1000).toISOString();
  const listings: SellerListing[] = []; let pages = 1;
  for (let page = 1; page <= pages && page <= 30; page += 1) { const result = await sellerPage(token, page, from, to); pages = result.totalPages; listings.push(...result.items); }
  return { listings, pages };
}

async function inventoryImages(token: string, sku: string) {
  if (!sku) return { status: 0, images: [] as string[], error: "Seller listing has no SKU" };
  const response = await fetch(`https://api.ebay.com/sell/inventory/v1/inventory_item/${encodeURIComponent(sku)}`, { headers: { Authorization: `Bearer ${token}`, "Content-Language": "en-US" }, signal: AbortSignal.timeout(30000) });
  const payload = await response.json().catch(() => ({})); return { status: response.status, images: response.ok && Array.isArray(payload?.product?.imageUrls) ? normalize(payload.product.imageUrls) : [], error: response.ok ? null : JSON.stringify(payload).slice(0, 500) };
}

async function rewrite(supabase: ReturnType<typeof createSupabaseServerClient>, inventory: InventoryRow, existing: ImageRow[], remote: string[]) {
  const current = normalize(existing.slice().sort((a,b) => a.sort_order - b.sort_order).map(x => x.image_url)); const finalImages = normalize([...remote, ...current]); const changed = current.length !== finalImages.length || current.some((url, i) => identity(url) !== identity(finalImages[i])); if (!changed || !finalImages.length) return { finalImages, changed: false };
  const { error: del } = await supabase.from("inventory_images").delete().eq("inventory_item_id", inventory.id); if (del) throw del;
  const { error: ins } = await supabase.from("inventory_images").insert(finalImages.map((image_url, i) => ({ inventory_item_id: inventory.id, image_url, alt_text: `${inventory.title} ${i === 0 ? "front" : i === 1 ? "back" : `detail ${i + 1}`}`, sort_order: i, is_primary: i === 0 }))); if (ins) throw ins; return { finalImages, changed: true };
}

export async function POST(request: Request) {
  const expected = process.env.EBAY_SELLER_SKU_IMAGE_REPAIR_TOKEN_20260730 || ""; const supplied = request.headers.get("x-tcos-repair-token") || ""; if (!expected || !safeEqual(expected, supplied)) return Response.json({ error: "Unauthorized" }, { status: 401 });
  const supabase = createSupabaseServerClient({ admin: true }); const token = await accessToken(supabase); const seller = await allSellerListings(token); const sellerById = new Map(seller.listings.map(x => [x.itemId, x]));
  const { data: productData, error: productError } = await supabase.from("products").select("id,title,ebay_item_id").eq("store_id", STORE_ID).in("id", TARGET_IDS).order("id"); if (productError) throw productError; const products = (productData || []) as ProductRow[];
  const { data: inventoryData, error: inventoryError } = await supabase.from("inventory_items").select("id,legacy_product_id,title").eq("store_id", STORE_ID).in("legacy_product_id", TARGET_IDS).order("legacy_product_id"); if (inventoryError) throw inventoryError; const inventories = (inventoryData || []) as InventoryRow[];
  const imageRows: ImageRow[] = []; for (let i = 0; i < inventories.length; i += 20) { const ids = inventories.slice(i, i + 20).map(x => x.id); const { data, error } = await supabase.from("inventory_images").select("id,inventory_item_id,image_url,sort_order,is_primary").in("inventory_item_id", ids).order("inventory_item_id").order("sort_order"); if (error) throw error; imageRows.push(...((data || []) as ImageRow[])); }
  const invByProduct = new Map<number, InventoryRow[]>(); for (const inv of inventories) { const list = invByProduct.get(Number(inv.legacy_product_id)) || []; list.push(inv); invByProduct.set(Number(inv.legacy_product_id), list); } const imgsByInv = new Map<string, ImageRow[]>(); for (const image of imageRows) { const list = imgsByInv.get(image.inventory_item_id) || []; list.push(image); imgsByInv.set(image.inventory_item_id, list); }
  const results: Array<Record<string, unknown>> = [];
  await workers(products, async product => { try { const listing = sellerById.get(String(product.ebay_item_id || "")); const api = await inventoryImages(token, listing?.sku || ""); let remote = normalize([...(api.images || []), ...(listing?.pictureUrls || [])]); if (product.id === 1992) remote = normalize([...remote, VERIFIED_1992_BACK]); const linked = invByProduct.get(product.id) || []; const finalSets: string[][] = []; let changedRows = 0; for (const inv of linked) { const r = await rewrite(supabase, inv, imgsByInv.get(inv.id) || [], remote); finalSets.push(r.finalImages); if (r.changed) changedRows += 1; } const combined = normalize([...remote, ...finalSets.flat()]); if (combined[0]) { const { error } = await supabase.from("products").update({ image_url: combined[0] }).eq("store_id", STORE_ID).eq("id", product.id); if (error) throw error; } results.push({ productId: product.id, title: product.title, ebayItemId: product.ebay_item_id, sellerListingFound: Boolean(listing), sellerSku: listing?.sku || null, sellerPictureCount: listing?.pictureUrls.length || 0, inventoryApiStatus: api.status, inventoryApiCount: api.images.length, inventoryApiError: api.error, finalImageCount: combined.length, changedRows, hasBack: product.id === 1992 || api.images.length >= 2 || (listing?.pictureUrls.length || 0) >= 2 }); } catch (error) { results.push({ productId: product.id, title: product.title, error: error instanceof Error ? error.message.slice(0,500) : String(error), hasBack: false }); } });
  results.sort((a,b) => Number(a.productId)-Number(b.productId)); const stillMissing = results.filter(x => x.hasBack !== true).map(x => ({ productId: x.productId, title: x.title, sellerSku: x.sellerSku || null, inventoryApiStatus: x.inventoryApiStatus || null, inventoryApiError: x.inventoryApiError || null, error: x.error || null }));
  return Response.json({ success: true, event: "ebay_seller_sku_image_repair_completed", sellerPages: seller.pages, sellerListings: seller.listings.length, requested: TARGET_IDS.length, foundProducts: products.length, repaired: results.filter(x => x.hasBack === true).length, stillMissingCount: stillMissing.length, stillMissing, results }, { headers: { "Cache-Control": "no-store" } });
}
